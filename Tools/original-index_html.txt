    <script>
        const vacations = {
            "maritimes-2026": {
                title: "2026 Maritimes",
                slides: [
                //  DELETE THIS AND THE NEXT TWO LINES AND PASTE ALL THE LINES FROM THE FILE CREATED BY THE "run-generate-album-list.cmd"
                    { src: "2026 Maritimes/Aug 7/1-Saint John, NB/01.jpg", day: "Aug 7 — Saint John, NB", caption: "Welcome to Saint John!" },
                    { src: "2026 Maritimes/Aug 7/2-Sawmill Creek Bridge, NB/01.jpg", day: "Aug 7 — Sawmill Creek Bridge, NB", caption: "The covered bridge lookouts." }
                 ]
            }
        };

        let currentActiveSlides = [];
        let currentIndex = 0;
        const captionStorageKey = 'maritimes-slideshow-captions-v1';
        let savedCaptions = {};

        function albumCommentsPath(slides) {
            const albumFolderName = slides[0]?.src.split('/')[0] || 'photo-albums';
            const albumFileName = albumFolderName.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
            return `${albumFolderName}/${albumFileName}-comments.json`;
        }

        function currentCaption(slide) {
            return savedCaptions[slide.src] ?? slide.caption ?? '';
        }

        async function loadCaptions(slides) {
            savedCaptions = {};
            try {
                const response = await fetch(encodeURI(albumCommentsPath(slides)));
                if (!response.ok) throw new Error('Comments file not found');
                const comments = await response.json();
                if (comments && !Array.isArray(comments) && typeof comments === 'object') savedCaptions = comments;
            } catch (error) {
                try {
                    savedCaptions = JSON.parse(localStorage.getItem(captionStorageKey)) || {};
                } catch (storageError) {
                    savedCaptions = {};
                }
            }
        }

        function initMenu() {
            const menuGrid = document.getElementById('menu-options');
            menuGrid.innerHTML = '';
            Object.keys(vacations).forEach(key => {
                const btn = document.createElement('button');
                btn.className = 'menu-btn';
                btn.innerText = vacations[key].title;
                btn.onclick = () => startSlideshow(key);
                menuGrid.appendChild(btn);
            });
        }

        async function startSlideshow(vacationKey) {
            currentActiveSlides = vacations[vacationKey].slides;
            currentIndex = 0;
            document.getElementById('menu-screen').classList.add('hidden');
            document.getElementById('viewer-screen').classList.remove('hidden');
            await loadCaptions(currentActiveSlides);
            displayMedia();
        }

        function displayMedia() {
            if (!currentActiveSlides.length) return;
            const slide = currentActiveSlides[currentIndex];
            const imgTag = document.getElementById('main-photo');
            const videoTag = document.getElementById('main-video');
            videoTag.pause();

            if (slide.src.toLowerCase().endsWith('.mp4')) {
                imgTag.classList.add('hidden');
                videoTag.classList.remove('hidden');
                videoTag.src = slide.src;
            } else {
                videoTag.classList.add('hidden');
                imgTag.classList.remove('hidden');
                imgTag.src = slide.src;
            }

            const dayParts = slide.day.split(' — ');
            document.getElementById('date-label').innerText = dayParts.shift();
            document.getElementById('folder-label').innerText = dayParts.join(' — ');
            document.getElementById('filename-label').innerText = slide.src.split('/').pop();
            document.getElementById('caption-label').innerText = currentCaption(slide);
            document.getElementById('photo-counter').innerText = `${currentIndex + 1} / ${currentActiveSlides.length}`;
        }

        function navigate(direction) {
            if (!currentActiveSlides.length) return;
            currentIndex += direction;
            if (currentIndex >= currentActiveSlides.length) currentIndex = 0;
            if (currentIndex < 0) currentIndex = currentActiveSlides.length - 1;
            displayMedia();
        }

        function goHome() {
            document.getElementById('main-video').pause();
            document.getElementById('viewer-screen').classList.add('hidden');
            document.getElementById('menu-screen').classList.remove('hidden');
        }

        document.addEventListener('keydown', (e) => {
            if (document.getElementById('viewer-screen').classList.contains('hidden')) return;
            if (e.key === 'ArrowRight') navigate(1);
            if (e.key === 'ArrowLeft') navigate(-1);
            if (e.key === 'Escape') goHome();
        });

        initMenu();
    </script>
